void runMainMenu(Settings *settings, MenuState *menuState) {
    for(int i = 0; i < 4; i++) {
        if(isMouseOnButton(&menuState->mainMenuButtons[i])) {
            highlightButton(&menuState->mainMenuButtons[i]);
            if(IsMouseButtonPressed(MOUSE_BUTTON_LEFT)) {
                switch(i) {
                    case 0:
                        menuState->menu = fullExit;
                        break;
                    case 1:
                        menuState->menu = playGame;
                        break;
                    case 2:
                        menuState->menu = onlineMenu;
                        break;
                    case 3:
                        menuState->menu = settingsMenu;
                        break;
                }
            }
        }
        else if(menuState->mainMenuButtons[i].rect.x < 600 && menuState->mainMenuButtons[i].extendLeft) {
            menuState->mainMenuButtons[i].rect.x+=20;
            menuState->mainMenuButtons[i].textPos.x+=20;
        }
        drawButton(settings, &menuState->mainMenuButtons[i]);
    }
}

void runOnlineMenu(Settings *settings, MenuState *menuState) {
    for(int i = 0; i < 1; i++) {
        if(isMouseOnButton(&menuState->settingsMenuButtons[i])) {
            highlightButton(&menuState->settingsMenuButtons[i]);
            if(IsMouseButtonPressed(MOUSE_BUTTON_LEFT)) {
                switch(i) {
                    case 0:

                        break;

            }
        }

            menuState->settingsMenuButtons[i].rect.x+=20;
            menuState->settingsMenuButtons[i].textPos.x+=20;
        }
        drawButton(settings, &menuState->settingsMenuButtons[i]);

}

void runSettingsMenu(Settings *settings, MenuState *menuState) {
    for(int i = 0; i < 2; i++) {
        if(isMouseOnButton(&menuState->settingsMenuButtons[i])) {
            highlightButton(&menuState->settingsMenuButtons[i]);
            if(IsMouseButtonPressed(MOUSE_BUTTON_LEFT)) {
                switch(i) {
                    case 0:
                        menuState->menu = mainMenu;
                        break;
                    case 1:
                        saveSettings(settings, "data/settings.txt");
                        break;
                }
            }
        }
        else if(menuState->settingsMenuButtons[i].rect.x < 600 && menuState->settingsMenuButtons[i].extendLeft) {
            menuState->settingsMenuButtons[i].rect.x+=20;
            menuState->settingsMenuButtons[i].textPos.x+=20;
        }
        drawButton(settings, &menuState->settingsMenuButtons[i]);
    }
}

MenuState *createMenuState(Settings *settings) {
    MenuState *menuState = malloc(sizeof(MenuState));
    menuState->mainMenuButtons = malloc(sizeof(MenuButton) * 4);
    menuState->onlineMenuButtons = malloc(sizeof(MenuButton) * 1);
    menuState->settingsMenuButtons = malloc(sizeof(MenuButton) * 2);

    menuState->mainMenuButtons[0] = createExitButton(settings);
    menuState->mainMenuButtons[1] = createSoloButton(settings);
    menuState->mainMenuButtons[2] = createOnlineButton(settings);
    menuState->mainMenuButtons[3] = createSettingsButton(settings);
    
    menuState->onlineMenuButtons[0] = createExitButton(settings);
    

    menuState->settingsMenuButtons[0] = createExitButton(settings);
    menuState->settingsMenuButtons[1] = createSaveButton(settings);

    return menuState;
}

    button.rect.height = 50;

    button.font = GetFontDefault(); // Use a custom font later, smth like roboto or smth
    button.fontSize = 10;
    button.textColor = RED;
    button.text = "Back";
    button.spacing = 5;

    return button;
}

void drawButton(Settings *settings, MenuButton *button) {
    DrawRectangleRec(button->rect, button->color);
    DrawRectangleLinesEx(button->rect, settings->OutlineThickness, button->textColor);
    DrawTextEx(
        button->font,
        button->text,
        button->textPos,
        button->fontSize,
        button->spacing,
        button->textColor
    );
}

bool isMouseOnButton(MenuButton *button) {
    if(
        GetMouseX() > button->rect.x &&
        GetMouseX() < button->rect.x + button->rect.width &&
        GetMouseY() > button->rect.y &&
        GetMouseY() < button->rect.y + button->rect.height
) return true;
return false;
}

void highlightButton(MenuButton *button) {
    DrawRectangleRec(button->rect, TRANSPARENT_WHITE);
    if(button->rect.x > 400 && button->extendLeft) {
        button->rect.x-=20;
        button->textPos.x-=20;
    }
}

void freeMenuState(MenuState *menuState) {
    free(menuState->mainMenuButtons);
    free(menuState->onlineMenuButtons);
    free(menuState->settingsMenuButtons);
    free(menuState);
}
